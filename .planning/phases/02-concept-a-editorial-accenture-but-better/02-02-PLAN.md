- Write the link-integrity check below into the verify step and make it pass. It resolves every local href across concept-a: absolute `/...` paths must map to existing files under the repo root (directories resolve via index.html), and every fragment (`#id`, standalone or after a path) must exist as an `id` in the target file. External http(s) hrefs are skipped.

```bash
python3 - <<'EOF'
import re, sys
from pathlib import Path
root = Path('.')
pages = [Path('concept-a/index.html'), *sorted(Path('concept-a/pages').glob('*.html'))]
ids = {}
def ids_of(p):
    if p not in ids: ids[p] = set(re.findall(r'id="([^"]+)"', p.read_text()))
    return ids[p]
fails = []
for page in pages:
    for href in re.findall(r'href="([^"]+)"', page.read_text()):
        if href.startswith(('http://', 'https://', 'mailto:')) or href.startswith('//'): continue
        path_part, _, frag = href.partition('#')
        if path_part == '': target = page
        else:
            t = root / path_part.lstrip('/') if path_part.startswith('/') else (page.parent / path_part)
            if t.is_dir(): t = t / 'index.html'
            if not t.exists(): fails.append(f'{page}: missing target {href}'); continue
            target = t.resolve().relative_to(root.resolve())
        if frag and frag not in ids_of(Path(target)): fails.append(f'{page}: missing anchor #{frag} in {target}')
print('\n'.join(fails) if fails else f'link-integrity OK across {len(pages)} pages')
sys.exit(1 if fails else 0)
EOF
```
