import json
hp = json.load(open('content/homepage.json'))
sp = json.load(open('content/subpages.json'))
def resolve(path):
    cur = hp
    for part in path.split('.'):
        cur = cur[int(part)] if isinstance(cur, list) else cur[part]
    return cur
def walk(node):
    if isinstance(node, dict):
        for k, v in node.items():
            if k.endswith('_ref'): resolve(v)
            elif k.endswith('_refs'):
                for r in v: resolve(r)
            else: walk(v)
    elif isinstance(node, list):
        for v in node: walk(v)
